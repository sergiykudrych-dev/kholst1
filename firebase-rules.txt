ПРАВИЛА БАЗЫ FIREBASE (Realtime Database → Rules)
==================================================

ПОРЯДОК ДЛЯ ДРУЗЕЙ (обновление):
 1. Сначала вставьте правила ниже (они включают и старые, так что сайт продолжит
    работать) и нажмите Publish.
 2. Потом загрузите новые файлы сайта на GitHub.

Что добавилось: codes (коды друзей), friends (списки друзей), invites
(приглашения в игру), history (история партий для статистики), ads (рекламные баннеры; пишет только владелец), stats (счётчики для панели владельца;
замените ВАША_ПОЧТА@gmail.com на почту вашего Google-аккаунта). Анонимный вход должен быть включён
(Authentication → Sign-in method → Anonymous).


ВАРИАНТ 1 (рекомендую) — «только со входом»
--------------------------------------------
{
  "rules": {
    "users": {
      "$uid": {
        ".read": "auth != null",
        ".write": "auth != null && auth.uid === $uid"
      }
    },
    "codes": {
      "$code": {
        ".read": "auth != null",
        ".write": "auth != null && !data.exists() && newData.val() === auth.uid"
      }
    },
    "friends": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        "$fid": {
          ".write": "auth != null && (auth.uid === $uid || (auth.uid === $fid && (!newData.exists() || newData.child('status').val() === 'in' || (newData.child('status').val() === 'ok' && data.child('status').val() === 'out'))))"
        }
      }
    },
    "invites": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        "$id": {
          ".write": "auth != null && (auth.uid === $uid || root.child('friends').child($uid).child(auth.uid).child('status').val() === 'ok')"
        }
      }
    },
    "push": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && auth.uid === $uid"
      }
    },
    "history": {
      "$uid": {
        ".read": "auth != null && auth.uid === $uid",
        ".write": "auth != null && auth.uid === $uid",
        "$id": {
          ".validate": "newData.hasChildren(['game', 'res', 't']) && (newData.child('res').val() === 'win' || newData.child('res').val() === 'loss' || newData.child('res').val() === 'draw' || newData.child('res').val() === 'coop')"
        }
      }
    },
    "stats": {
      ".read": "auth != null && auth.token.email === 'ВАША_ПОЧТА@gmail.com' && auth.token.email_verified === true",
      ".write": "auth != null && auth.token.email === 'ВАША_ПОЧТА@gmail.com' && auth.token.email_verified === true",
      "daily": {
        "$day": {
          "$key": {
            ".write": "auth != null && newData.isNumber() && newData.val() === (data.exists() ? data.val() : 0) + 1"
          }
        }
      },
      "total": {
        "users": {
          ".write": "auth != null && newData.isNumber() && newData.val() === (data.exists() ? data.val() : 0) + 1"
        }
      },
      "users": {
        "$day": {
          "$uid": {
            ".write": "auth != null && auth.uid === $uid && !data.exists() && newData.val() === 1"
          }
        }
      },
      "ads": {
        "$id": {
          "$k": {
            ".write": "auth != null && ($k === 'imp' || $k === 'clk') && newData.isNumber() && newData.val() === (data.exists() ? data.val() : 0) + 1"
          }
        }
      },
      "seen": {
        "$uid": {
          ".write": "auth != null && auth.uid === $uid && !data.exists() && newData.val() === 1"
        }
      }
    },
    "ads": {
      ".read": "auth != null",
      ".write": "auth != null && auth.token.email === 'ВАША_ПОЧТА@gmail.com' && auth.token.email_verified === true"
    },
    "rooms": {
      "$room": {
        ".read": "auth != null",
        ".write": "auth != null"
      }
    }
  }
}
